// ===================================================== // (주)솔루션 전자책 — Google Drive 자동 동기화 스크립트 // ===================================================== // 설정 방법: // 1. script.google.com 접속 → 새 프로젝트 // 2. 이 코드 전체 붙여넣기 // 3. 아래 CONFIG 값 채우기 (토큰만 직접 입력) // 4. 상단 메뉴 → 트리거 → syncNewPDFs 함수를 1시간마다 실행 // ===================================================== const CONFIG = { DRIVE_FOLDER_ID: '1tg_LQn_O16LeYHpSbzO4DwDEiIl122NL', // PDF 보관 Drive 폴더 GITHUB_OWNER: 'soullution21-stack', GITHUB_REPO: 'solution-casebook', GITHUB_BRANCH: 'main', GITHUB_TOKEN: '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', // ← 여기에 본인 토큰 입력 }; // ===================================================== // 메인 함수 — 트리거로 주기적으로 실행됨 // ===================================================== function syncNewPDFs() { const folder = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID); const files = folder.getFilesByType(MimeType.PDF); // GitHub에서 현재 index.html 가져오기 const { content: indexHtml, sha: indexSha } = getGithubFile('index.html'); // 이미 등록된 Drive 파일 ID 목록 추출 const registeredIds = extractRegisteredDriveIds(indexHtml); let updated = false; let newIndexHtml = indexHtml; while (files.hasNext()) { const file = files.next(); const fileName = file.getName(); // 예: 2026_스마트생태공장_주_예시업체_68.pdf // PDF 파일명 형식 검증 if (!fileName.endsWith('.pdf')) continue; const baseName = fileName.replace('.pdf', ''); if (!isValidFileName(baseName)) { Logger.log(`⚠️ 파일명 형식 불일치, 건너뜀: ${fileName}`); continue; } const driveId = file.getId(); // 이미 등록된 파일이면 건너뜀 if (registeredIds.includes(driveId)) { Logger.log(`✅ 이미 등록됨: ${fileName}`); continue; } Logger.log(`🆕 새 파일 발견: ${fileName}`); // 1. 커버 이미지 생성 (PDF 첫 페이지 → JPG) const coverSuccess = generateAndUploadCover(file, baseName); if (!coverSuccess) { Logger.log(`❌ 커버 생성 실패: ${fileName}`); continue; } // 2. index.html에 새 카드 데이터 추가 const cardJson = buildCardJson(baseName, driveId); newIndexHtml = insertCardToHtml(newIndexHtml, cardJson); updated = true; Logger.log(`✅ 추가 완료: ${fileName}`); } // 변경사항이 있으면 index.html GitHub에 업로드 if (updated) { uploadToGithub('index.html', newIndexHtml, indexSha, '자동 동기화: 새 사례 추가'); Logger.log('🚀 index.html GitHub 업로드 완료 → Netlify 자동 배포 시작'); } else { Logger.log('변경사항 없음'); } } // ===================================================== // 파일명 형식 검증 // 패턴: {연도}_{사업명}_{업체명}_{순번} // ===================================================== function isValidFileName(baseName) { // 연도로 시작하고 마지막이 숫자(순번)인지 확인 return /^\d{4}_.+_\d+$/.test(baseName); } // ===================================================== // 파일명에서 메타데이터 파싱 // ===================================================== function parseFileName(baseName) { const parts = baseName.split('_'); const year = parts[0]; const seq = parts[parts.length - 1]; // 사업명 파싱 (두 번째 토큰 기준) const businessMap = { '스마트생태공장': { short: '스마트생태공장', full: '스마트생태공장' }, '상생협력': { short: '상생협력 실증', full: '상생협력 실증' }, '탄소중립': { short: '탄소중립 전환지원', full: '탄소중립 전환지원' }, '탄소중립설비투자': { short: '탄소중립 설비투자 지원사업', full: '탄소중립 설비투자 지원사업' }, }; let businessShort = parts[1] || ''; let project = businessShort; for (const [key, val] of Object.entries(businessMap)) { if (baseName.includes(key)) { businessShort = val.short; project = val.full; break; } } // 업체명: 연도와 사업명, 순번 제외한 나머지 // 파일명에서 연도(parts[0])와 마지막 순번(parts[-1]) 제거 // 사업명 토큰 길이를 동적으로 처리하기 위해 인덱스 계산 let businessTokenCount = 1; if (baseName.includes('상생협력_실증')) businessTokenCount = 2; if (baseName.includes('탄소중립_전환지원')) businessTokenCount = 2; if (baseName.includes('탄소중립_설비투자_지원사업')) businessTokenCount = 3; const companyParts = parts.slice(1 + businessTokenCount, parts.length - 1); const company = companyParts.join('_').replace(/^주_/, '(주)').replace(/^주식회사_/, '주식회사 '); return { year, seq, businessShort, project, company }; } // ===================================================== // 카드 JSON 생성 // ===================================================== function buildCardJson(baseName, driveId) { const { year, seq, businessShort, project, company } = parseFileName(baseName); return { id: baseName, year: year, businessShort: businessShort, project: project, company: company, industry: '', // 관리모드에서 나중에 입력 assignment: '', // 관리모드에서 나중에 입력 cover: `assets/covers/${baseName}_cover.jpg`, pdf: `https://drive.google.com/file/d/${driveId}/preview`, pageCount: 1, thumbPage: 1, mainPage: 1, items: [], showCard: true, sourcePath: '', }; } // ===================================================== // index.html의 DEFAULT_DATA 배열에 카드 삽입 // ===================================================== function insertCardToHtml(html, cardJson) { // DEFAULT_DATA 배열의 마지막 항목 뒤에 삽입 const marker = /(\s*\]\s*;\s*\/\/\s*END_DEFAULT_DATA)/; const jsonStr = '\n ' + JSON.stringify(cardJson, null, 2).replace(/\n/g, '\n ') + ','; if (marker.test(html)) { return html.replace(marker, jsonStr + '$1'); } // marker가 없으면 DEFAULT_DATA = [ ... ] 패턴에서 마지막 } 앞에 삽입 // 배열 끝 ]을 찾아서 그 앞에 추가 const arrayEndPattern = /(const DEFAULT_DATA\s*=\s*\[[\s\S]*?)(\s*\]\s*;)/; if (arrayEndPattern.test(html)) { return html.replace(arrayEndPattern, '$1,\n ' + JSON.stringify(cardJson) + '$2'); } Logger.log('❌ DEFAULT_DATA 삽입 위치를 찾지 못했습니다. 수동으로 추가해주세요.'); return html; } // ===================================================== // 이미 등록된 Drive 파일 ID 추출 // ===================================================== function extractRegisteredDriveIds(html) { const matches = html.match(/drive\.google\.com\/file\/d\/([^/'"]+)/g) || []; return matches.map(m => m.replace('drive.google.com/file/d/', '')); } // ===================================================== // PDF 첫 페이지를 JPG로 변환하여 GitHub에 업로드 // ===================================================== function generateAndUploadCover(file, baseName) { try { // Google Drive에서 PDF를 Google Slides로 변환 (첫 페이지 이미지 추출용) const pdfBlob = file.getBlob(); // PDF → 썸네일 이미지 추출 // Drive API를 통해 파일 썸네일 가져오기 const fileId = file.getId(); const thumbnailUrl = `https://drive.google.com/thumbnail?id=${fileId}&sz=w800`; const response = UrlFetchApp.fetch(thumbnailUrl, { headers: { Authorization: `Bearer ${ScriptApp.getOAuthToken()}` }, muteHttpExceptions: true, }); if (response.getResponseCode() !== 200) { Logger.log(`썸네일 fetch 실패: ${response.getResponseCode()}`); return false; } const imageBlob = response.getBlob(); const base64Image = Utilities.base64Encode(imageBlob.getBytes()); // GitHub assets/covers/ 에 업로드 const coverPath = `assets/covers/${baseName}_cover.jpg`; const existingSha = getGithubFileSha(coverPath); // 이미 있으면 sha 필요 uploadToGithub(coverPath, base64Image, existingSha, `커버 이미지 추가: ${baseName}`, true); Logger.log(`📸 커버 업로드 완료: ${coverPath}`); return true; } catch (e) { Logger.log(`커버 생성 오류: ${e.message}`); return false; } } // ===================================================== // GitHub API — 파일 가져오기 // ===================================================== function getGithubFile(path) { const url = `https://api.github.com/repos/${CONFIG.GITHUB_OWNER}/${CONFIG.GITHUB_REPO}/contents/${path}?ref=${CONFIG.GITHUB_BRANCH}`; const res = UrlFetchApp.fetch(url, { headers: { Authorization: `token ${CONFIG.GITHUB_TOKEN}`, Accept: 'application/vnd.github.v3+json', }, muteHttpExceptions: true, }); const json = JSON.parse(res.getContentText()); const content = Utilities.newBlob(Utilities.base64Decode(json.content.replace(/\n/g, ''))).getDataAsString(); return { content, sha: json.sha }; } // ===================================================== // GitHub API — 파일 SHA만 가져오기 (없으면 null) // ===================================================== function getGithubFileSha(path) { try { const url = `https://api.github.com/repos/${CONFIG.GITHUB_OWNER}/${CONFIG.GITHUB_REPO}/contents/${path}?ref=${CONFIG.GITHUB_BRANCH}`; const res = UrlFetchApp.fetch(url, { headers: { Authorization: `token ${CONFIG.GITHUB_TOKEN}`, Accept: 'application/vnd.github.v3+json', }, muteHttpExceptions: true, }); if (res.getResponseCode() === 200) { return JSON.parse(res.getContentText()).sha; } return null; } catch (e) { return null; } } // ===================================================== // GitHub API — 파일 업로드/수정 // ===================================================== function uploadToGithub(path, content, sha, message, isBase64 = false) { const url = `https://api.github.com/repos/${CONFIG.GITHUB_OWNER}/${CONFIG.GITHUB_REPO}/contents/${path}`; const body = { message: message, branch: CONFIG.GITHUB_BRANCH, content: isBase64 ? content : Utilities.base64Encode(content), }; if (sha) body.sha = sha; const res = UrlFetchApp.fetch(url, { method: 'PUT', headers: { Authorization: `token ${CONFIG.GITHUB_TOKEN}`, Accept: 'application/vnd.github.v3+json', 'Content-Type': 'application/json', }, payload: JSON.stringify(body), muteHttpExceptions: true, }); const code = res.getResponseCode(); if (code !== 200 && code !== 201) { Logger.log(`GitHub 업로드 실패 (${code}): ${res.getContentText()}`); throw new Error(`GitHub 업로드 실패: ${path}`); } }
